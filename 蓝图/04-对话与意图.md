## Pydantic Schema

```python
class ConversationCreate(BaseModel):
    title: str | None = None

class ConversationResponse(BaseModel):
    id: UUID
    title: str
    created_at: datetime

class MessageCreate(BaseModel):
    content: str = Field(min_length=1, max_length=8000)
    image_id: UUID | None = None     # 附带图片（图片库引用，图片理解）

class MessageResponse(BaseModel):
    id: UUID
    role: str
    content: str
    tool_name: str | None
    attachments: list[Attachment] = []  # 历史图片引用（读取时后端补短期 URL）
    created_at: datetime

class IntentResult(BaseModel):
    """意图识别结构化输出。"""
    intent_name: str                    # 见 INTENTS 表
    parameters: dict[str, Any]          # 实体/参数；路由后按意图 schema 二次校验（见下表）
    confidence: float = Field(ge=0, le=1)

class ChatRequest(BaseModel):
    conversation_id: UUID
    content: str

class Attachment(BaseModel):
    """问答/消息里的图片或文件引用。url 已带短期展示 token，前端直接 <img>。"""
    type: Literal["image", "file"] = "image"
    image_id: UUID | None = None
    url: str | None = None
    thumbnail_url: str | None = None
    filename: str | None = None

class ChatResponse(BaseModel):
    reply: str
    intent: IntentResult
    tool_calls: list[str] = []          # 本次调用的工具名
    attachments: list[Attachment] = []  # 图片/文件引用（图库检索/工具结果）
```

### agent/graph.py（主对话图）

```python
def build_main_graph() -> CompiledStateGraph:
    """intent_route → (工具执行 / 记忆 / RAG / Agent 子图) → finalize。"""
    g = StateGraph(AgentState)
    g.add_node("intent_route", intent_route)
    g.add_node("tool_executor", tool_executor)       # 11 定义，按 intent 调工具
    g.add_node("memory_write", memory_write)          # 05 定义
    g.add_node("memory_query", memory_query)          # 05 定义
    g.add_node("rag_retrieve", rag_retrieve)          # 06 定义
    g.add_node("skill_entry", skill_entry)            # 10 定义，跑 Skill 子图
    g.add_node("agent_selector", agent_selector)      # 04 定义，按复杂度选子图
    g.add_node("react_loop", build_react_subgraph(registry))       # 04 Agent 子图
    g.add_node("plan_loop", build_plan_subgraph(registry))         # 04 Agent 子图
    g.add_node("reflection_loop", build_reflection_subgraph(llm))  # 04 Agent 子图
    g.add_node("finalize", finalize)                  # 组装 memory/rag 上下文 + LLM 生成回复
    g.add_edge(START, "intent_route")
    g.add_conditional_edges("intent_route", route_by_intent, {
        "tool": "agent_selector", "memory": "memory_query",
        "memory_write": "memory_write", "rag": "rag_retrieve",
        "skill": "skill_entry",                       # 10 子图入口
        "chat": "finalize"})
    g.add_conditional_edges("agent_selector", agent_selector, {
        "tool": "tool_executor", "react": "react_loop",
        "plan": "plan_loop", "reflection": "reflection_loop",
        "chat": "finalize"})
    # tool_executor / skill_entry / 各 loop → memory_write（记住本 turn）→ finalize
    # memory_query / rag_retrieve → finalize
    g.add_edge("tool_executor", "memory_write")
    g.add_edge("skill_entry", "memory_write")
    g.add_edge("react_loop", "memory_write")
    g.add_edge("plan_loop", "memory_write")
    g.add_edge("reflection_loop", "memory_write")
    g.add_edge("memory_write", "finalize")
    g.add_edge("finalize", END)
    return g.compile(checkpointer=RedisSaver(redis), interrupt_before=[...])
```

### 三种 Agent 子图（agent/loops/）

一个 Agent = 一个编译过的小图（自己的 loop），作为主图节点挂载。共用主图 AgentState、ToolRegistry（11）、checkpointer。参考 zhao/agent/ 的 ReAct/PlanExecute/Reflection 实现思路，用 LangGraph 子图重写。

#### ReAct 子图（loops/react.py）——多轮工具调用

```python
class ReActState(TypedDict, total=False):
    task: str
    user_id: UUID
    conversation_id: UUID
    steps: list[dict]            # [{thought, action, observation}, ...]
    final: str

def build_react_subgraph(registry) -> CompiledStateGraph:
    """thought → (act ↔ think 循环) → 答案。最大 8 轮防死循环。"""
```

节点伪代码：
```
think:   LLM 输入 = task + 历史 steps + ToolRegistry 工具列表
         → 结构化输出 {thought, action: {tool, params} | answer}
         action 是 answer → 写 final → END；是 tool → act
act:     ToolRegistry.call(user_id, tool, params) → observation
         （含 pending_confirmation 挂起，确认流程同 11）
         steps.append({thought, action, observation}) → 回 think
         轮数 ≥ 8 → 强制出答案
```

#### PlanExecute 子图（loops/plan_execute.py）——规划多步任务

```python
class PlanStep(BaseModel):
    step: str                    # 做什么
    tool: str | None = None      # 用哪个工具（空=LLM 直接做）
    params: dict = {}

class PlanExecuteState(TypedDict, total=False):
    task: str
    user_id: UUID
    conversation_id: UUID
    plan: list[PlanStep]
    idx: int
    results: list[dict]
    final: str

def build_plan_subgraph(registry) -> CompiledStateGraph:
    """plan → execute ↔ replan 循环。"""
```

节点伪代码：
```
plan:    LLM 生成 PlanStep[]（结构化输出，三层兜底）
execute: 执行 plan[idx]（tool 非空→registry.call；空→LLM 直接做）→ results.append
replan:  LLM 判断结果是否偏离/需调整 → 是则重生成 plan 复位 idx，否则 idx+1
循环:    idx < len(plan) 且未结束 → execute；结束 → 汇总 final
```

**演示场景（面试讲深点）——"规划周末旅行"**：
```
用户: 帮我规划周末去杭州的旅行
→ plan: [查天气, 搜景点, 排日程, 列清单, 设提醒]
→ execute: web_search(杭州天气) → web_search(杭州景点) → create_schedule(日程×N)
           → create_todo(清单) → schedule_reminder(出发提醒)
→ 展示: 多步工具链 + 工具确认（create_modify/send_delete 挂起→确认）+ 审计链
```

#### Reflection 子图（loops/reflection.py）——生成质量把关

```python
class ReflectionState(TypedDict, total=False):
    task: str
    user_id: UUID
    conversation_id: UUID
    draft: str
    iterations: int
    final: str

def build_reflection_subgraph(llm) -> CompiledStateGraph:
    """generate → critique ↔ revise，通过（score≥8）或满 3 轮结束。"""
```

节点伪代码：
```
generate: LLM 生成初稿
critique: LLM 评审 → 结构化输出 {score, issues[]}
          通过 或 iterations ≥ 3 → final = draft → END；否则 revise
revise:   按 issues 修订 → iterations+1 → 回 critique
```

`intent_route` 伪代码：
```
last = state["messages"][-1]["content"]
prompt = f"""识别用户意图与实体，只输出 JSON。
可选意图：{INTENTS 列表}。实体按参数 schema 提取。
用户：{last}"""
llm = INTENT_LLM.with_structured_output(IntentResult)
try:
    result = await llm.ainvoke(prompt)
    result.parameters = validate_intent_params(result)   # ② 意图参数 schema 二次校验
except Exception:
    result = IntentResult(intent_name="default_chat", parameters={}, confidence=0.0)
return {"intent": result}
```

`validate_intent_params(result)`：按 `INTENT_PARAM_SCHEMAS[intent_name]` 对 parameters 做 `model_validate`，失败返回默认/空参数并降级到该意图的可执行默认。

**结构化输出三层兜底**（对应《结构化输出与可控生成》三级）：
1. `with_structured_output` —— function calling / JSON mode 降级（DeepSeek/千问无 OpenAI 级约束解码，字段/类型可能漂移）
2. 服务端 Pydantic `model_validate` 二次校验 —— 必做，靠 schema 兜字段名和类型
3. 失败重试 N 次 → 仍失败走兜底分支（default_chat / 空参数）

### agent/runner.py（AgentRunner）

```python
class AgentRunner:
    """对话入口：跑图 + 持久化消息。全局单例。"""
    def __init__(self, graph, registry):
        self._graph = graph; self._registry = registry   # registry = ToolRegistry（11）

    def _thread(self, user_id: UUID, conversation_id: UUID) -> dict:
        return {"configurable": {"thread_id":
                redis_key("session", user_id, conversation_id)}}

    async def run(self, user_id: UUID, conversation_id: UUID,
                  content: str, image_id: UUID | None = None) -> ChatResponse:
        """调图 → 把 assistant/tool 消息 append 进 messages 表 → 返回 ChatResponse。
        image_id：消息附带图片（图片理解，VLM 描述注入）。"""

    async def get_history(self, user_id: UUID,
                          conversation_id: UUID) -> list[MessageResponse]:
        """从 messages 表读持久历史（业务数据，非 checkpointer）。"""

    async def stream(self, user_id: UUID, conversation_id: UUID,
                     content: str, image_id: UUID | None = None) -> AsyncIterator[str]:
        """同 run，但走 LangGraph astream，LLM 输出以 SSE chunk 流式返回（打字机效果）。
        API 层 `Accept: text/event-stream` 时启用；否则回退 run 非流式。"""
```

`run` 伪代码：
```
# 0. 本轮是"确认/取消"？直接处理挂起动作，不走意图路由
action = classify_confirm_input(content)    # "确认"→confirm / "取消"→deny
if action is not None:
    result = await self._registry.confirm_latest(user_id, conversation_id) if action == "confirm" \
             else await self._registry.deny_latest(user_id, conversation_id)
    reply = format_pending_result(result)
    await message_repo.create(db, conversation_id, user_id, "user", content)
    await message_repo.create(db, conversation_id, user_id, "assistant", reply)
    await db.commit()
    return ChatResponse(reply=reply, intent=IntentResult(intent_name="confirm", parameters={}, confidence=1.0))
# 0.5 消息带图片：VLM 生成图像描述注入上下文（图片理解）
if image_id:
    img_desc = await vision_llm.describe(image_id)   # Qwen-VL，01 vision_model
    content = f"[图片内容] {img_desc}\n{content}"
# 1. 持久化用户消息
await message_repo.create(db, conversation_id, user_id, "user", content)
# 2. 跑图
inputs = {"messages": [{"role": "user", "content": content}],
          "user_id": user_id, "conversation_id": conversation_id}
result = await self._graph.ainvoke(inputs, config=self._thread(...))
# 3. 持久化 assistant 回复（+ 工具调用记录走 11 audit）
await message_repo.create(db, conversation_id, user_id, "assistant", result["reply"])
await db.commit()
return ChatResponse(reply=result["reply"], intent=result["intent"],
                    tool_calls=[r["name"] for r in result.get("tool_results", [])])
```
（补充：本轮工具挂起（`status=pending_confirmation`）后，回复里提示"有一项待确认：xxx，回复『确认』执行"；
下一轮进入 run 时按确认/取消分支走 `registry.confirm_latest` / `deny_latest`（11），不重跑意图路由。
对话侧获取待确认列表：`self._registry.get_pending(user_id, conversation_id)`。）

图片引用：tool_results 中 `search_images`/`get_image`（06）的返回值 → `ChatResponse.attachments`；
消息持久化时只存 `image_id` 进 meta（不含 token），历史读取（get_messages）时后端按 image_id 补短期展示 URL——token 会过期，不能写死进库。

`classify_confirm_input(content)`：判定"确认/取消"（"确认/可以/好的"→confirm；"取消/不要/算了"→deny；否则 None）。
`format_pending_result(result)`：把 ToolResult（done/denied）转成用户可见回复文本。

## 测试要点

- 意图识别：create_schedule 类语句 → 参数含时间/标题
- 工具路由：识别结果正确映射到分支
- 多轮上下文：checkpointer 保留上轮（Redis 会话隔离）
- 消息持久化：turn 结束 messages 表有 user/assistant 两行
- 隔离：A 无法读 B 的会话
