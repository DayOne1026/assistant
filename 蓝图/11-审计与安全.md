## Pydantic Schema

```python
class ToolLevel(str, Enum):
    READ_ONLY = "read_only"          # 直接执行
    CREATE_MODIFY = "create_modify"  # 用户确认后执行
    SEND_DELETE = "send_delete"      # 二次确认
    HIGH_RISK = "high_risk"          # 禁止注册

class ToolResult(BaseModel):
    status: Literal["done", "pending_confirmation", "denied"]
    call_id: UUID | None = None
    data: Any = None
    message: str = ""

class AuditLogResponse(BaseModel):
    id: UUID
    action: str
    resource_type: str
    resource_id: UUID | None
    detail: dict | None
    created_at: datetime

class ToolLogResponse(BaseModel):
    id: UUID
    tool_name: str
    level: str
    decision: str
    created_at: datetime
```

## 工具权限框架（agent/tools.py）

```python
class ToolDef(BaseModel):
    name: str
    description: str
    parameters_schema: dict          # JSON Schema，供 LLM 生成参数 + 服务端校验
    level: ToolLevel
    handler: Callable[[UUID, dict], Awaitable[Any]]   # (user_id, params) -> result

class ToolRegistry:
    """工具注册表：注册/查找/调用（权限+确认+审计）。全局单例。"""
    def __init__(self, db, redis, audit): ...
    def register(self, tool: ToolDef) -> None:
        """HIGH_RISK 工具拒绝注册（抛异常）；重名拒绝。"""
    def get(self, name: str) -> ToolDef: ...
    def list(self) -> list[ToolDef]: ...
    async def call(self, user_id: UUID, name: str, params: dict,
                   conversation_id: UUID | None = None) -> ToolResult:
        """权限检查→参数校验→(直接执行 | 挂起待确认)→审计。"""
    async def confirm(self, user_id: UUID, call_id: UUID,
                      conversation_id: UUID | None = None) -> ToolResult:
        """精确确认；给了 conversation_id 则校验该 call 属于此会话（防串）。"""
    async def get_pending(self, user_id: UUID,
                          conversation_id: UUID | None) -> list[ToolResult]:
        """查该会话挂起的待确认动作（供下一轮展示给用户）。"""
    async def confirm_latest(self, user_id: UUID,
                             conversation_id: UUID) -> ToolResult:
        """确认当前会话最近一个挂起动作（对话里用户说"确认"时用）。"""
    async def deny(self, user_id: UUID, call_id: UUID) -> ToolResult:
        """拒绝待确认动作：审计置 denied，删 Redis。"""
    async def deny_latest(self, user_id: UUID,
                          conversation_id: UUID) -> ToolResult:
        """拒绝当前会话最近一个挂起动作。"""
```

`call` 伪代码：
```
tool = self.get(name)                        # 不存在 → AppException(NOT_FOUND)
if tool.level is ToolLevel.HIGH_RISK:
    await audit.log_tool_call(..., decision="denied"); return ToolResult(status="denied")
validate_params(params, tool.parameters_schema)   # 失败 → VALIDATION_ERROR
call_id = uuid4()
await audit.log_tool_call(user_id, conversation_id, name, params,
                          tool.level, "pending", call_id)
if tool.level is ToolLevel.READ_ONLY:
    data = await tool.handler(user_id, params)
    await audit.complete_tool_call(call_id, data)   # decision→approved
    return ToolResult(status="done", call_id=call_id, data=data)
# create_modify / send_delete：挂起
detail = json.dumps({"name": name, "params": params,
                     "conversation_id": str(conversation_id) if conversation_id else None})
await redis.set(redis_key("confirm-tool", user_id, str(call_id)), detail, ex=300)
if conversation_id:  # 按会话挂队列，支持"确认当前会话最近动作"
    await redis.rpush(redis_key("confirm-pending", user_id, str(conversation_id)),
                      str(call_id))
return ToolResult(status="pending_confirmation", call_id=call_id,
                  message="该操作需你确认")
```

`confirm` 伪代码：
```
raw = await redis.get(redis_key("confirm-tool", user_id, str(call_id)))
if raw is None: return ToolResult(status="denied", message="确认已过期或无效")
detail = json.loads(raw)
name, params = detail["name"], detail["params"]
if conversation_id and detail.get("conversation_id") != str(conversation_id):
    return ToolResult(status="denied", message="该动作不属于此会话")
tool = self.get(name)
data = await tool.handler(user_id, params)   # 执行前不再重查权限（执行对象已锁定）
await audit.complete_tool_call(call_id, data)
await redis.delete(redis_key("confirm-tool", user_id, str(call_id)))
if conversation_id:  # 纯 call_id 确认时队列无此项，跳过
    await redis.lrem(redis_key("confirm-pending", user_id, str(conversation_id)), 0, str(call_id))
return ToolResult(status="done", call_id=call_id, data=data)
```

`get_pending` 伪代码：
```
ids = await redis.lrange(redis_key("confirm-pending", user_id, str(conversation_id)), 0, -1)
result = []
for cid in ids:
    raw = await redis.get(redis_key("confirm-tool", user_id, cid))
    if raw is None: continue                 # 已过期，跳过
    d = json.loads(raw)
    result.append(ToolResult(status="pending_confirmation", call_id=UUID(cid),
                             message=f"待确认：{d['name']} {d['params']}"))
return result
```

`confirm_latest` 伪代码：

if not ids: return ToolResult(status="denied", message="无可确认动作")

## 中间件（middleware/）

### RateLimitMiddleware
```python
class RateLimitMiddleware(BaseHTTPMiddleware):
    async def dispatch(self, request, call_next):
        """Redis 滑动窗口（01）。未认证按 IP，已认证按 user_id+IP。
        超限 → 429 + RATE_LIMITED。豁免 /health /docs。"""
        key = user_id or ip   # user_id 从 Bearer token 解析（可选）；未认证仅按 IP
        allowed = await redis.sliding_window_rate_limit(
            f"app:ratelimit:{key}:{request.url.path}",
            settings.rate_limit_max, settings.rate_limit_window_seconds)
        if not allowed: return JSONResponse(429, fail(RATE_LIMITED, "请求过于频繁"))
        return await call_next(request)
```

### HarmfulFilterMiddleware
```python
class HarmfulFilterMiddleware(BaseHTTPMiddleware):
    async def dispatch(self, request, call_next):

        可选 LLM 分类（慢，默认关）。"""

### CORS / CSRF
- CORS：白名单来自 settings.cors_origins；允许方法 GET/POST/PATCH/DELETE；`Allow-Origin` 精确匹配，不用 `*`。
- CSRF：认证走 Bearer token（非 cookie），无跨站自动携带凭证面 → 主要防线是 CORS 白名单 + OAuth state（09）防跨站授权注入。

## 防提示注入（安全层，兜底）

1. **分隔符隔离**：system prompt 与用户内容用 `USER_CONTENT_SEP`（10）隔开，用户内容不进入指令区。
2. **用户 prompt 只追加**：自定义 System Prompt 只能在安全层后追加（10），不可覆盖安全指令。
3. **工具权限兜底**：即使提示词被污染，工具执行仍受 ToolLevel + 确认 + 审计三重约束，HIGH_RISK 不可注册。
4. **LLM 输出不直接执行**：工具参数经 JSON Schema 校验，handler 只接受结构化参数。

## 测试要点

- 四级权限行为：read_only 直接跑；create_modify/send_delete 挂起；high_risk 注册失败
- 确认流程：挂起→确认执行；过期拒绝；伪造 call_id 拒绝
- 参数校验：LLM 给出非法参数 → VALIDATION_ERROR
- 每次工具调用/写操作有审计记录，无遗漏
- 限流：超阈值 429；白名单豁免
- 审计隔离：A 查不到 B 的日志
